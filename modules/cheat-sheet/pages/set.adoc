////
.Parameters
[source, parameters]
----
{
  "value": "42"
}
----
////

====
[source, cypher, role=noheader]
----
MATCH (e:Label)
SET e.property1 = $value1
----

[.description]
Update or create a property.
====

////
.Parameters
[source, parameters]
----
{
  "value1": "43",
  "value2": "44"
}
----
////

====
[source, cypher, role=noheader]
----
MATCH (e:Label)
SET e.property1 = $value1,
    e.property2 = $value2
----

[.description]
Update or create several properties.
====

////
.Parameters
[source, parameters]
----
{
  "key": "propertyName",
  "value": "propertyValue"
}
----
////

====
[source, cypher, role=noheader]
----
MATCH (n:Label)
SET n[$key] = $value
----

[.description]
Dynamically set or update node properties.
====


////
.Parameters
[source, parameters]
----
{
  "label": "Label"
}
----
////

====
[source, cypher, role=noheader]
----
MATCH (n:Label)
SET n:$($label)
----

[.description]
Dynamically set node labels.
====


////
.Parameters
[source, parameters]
----
{
  { "map": { "prop1": "value1", "prop2": 42 } }
}
----
////

====
[source, cypher, role=noheader]
----
MATCH (e:Label)
SET e = $map
----

[.description]
Set all properties. This will remove any existing properties.
====


====
[source, cypher, role=noheader]
----
MATCH (e:Label)
SET e = {}
----

[.description]
Using the empty map (`{}`), removes any existing properties.
====


====
[source, cypher, role=noheader]
----
MATCH (e:Label)
SET e += $map
----

[.description]
Add and update properties, while keeping existing ones.
====


====
[source, cypher, role=noheader]
----
MATCH (n:Label)
WHERE n.id = 123
SET n:Person
----

[.description]
Add a label to a node. This example adds the label `Person` to a node.
====
